## 🎯 各科进度

```dataviewjs
const pages = dv.pages('"20-进度"').sort(p => p.file.name, 'asc');
for (const p of pages) {
  const prog = p.progress ?? 0;
  const filled = Math.round(prog / 5);
  const bar = "█".repeat(filled) + "░".repeat(20 - filled);
  dv.paragraph(`**${p.subject ?? p.file.name}** · ${p.phase ?? "未开始"}`);
  dv.paragraph(bar + ` ${prog}%`);
}
```
